#  ATENÇÃO — LEIA ANTES DE USAR
#
#  Este arquivo NÃO é enviado por padrão. Ele é opcional.
#
#  ⚠️  A raiz desta hospedagem também serve o `robo-compras.js` de OUTRO projeto.
#      Um .htaccess na raiz afeta esse arquivo também.
#
#  Antes de subir:
#    1. Verifique se já existe um .htaccess na raiz pelo Gerenciador de Arquivos
#       da Hostinger (ative "mostrar arquivos ocultos").
#    2. Se existir: NÃO sobrescreva. Copie apenas os blocos que você quer para
#       dentro do arquivo existente.
#    3. Se não existir: renomeie este arquivo para `.htaccess` e suba na raiz.
#
#  O site funciona perfeitamente SEM este arquivo. Ele só melhora cache e HTTPS.
#
#  Segurança: as regras de rewrite abaixo liberam qualquer arquivo que exista
#  de verdade no disco (RewriteCond %{REQUEST_FILENAME} !-f), então o
#  robo-compras.js continua sendo servido normalmente.
# =============================================================================


# ---- 1. Força HTTPS ---------------------------------------------------------
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTPS} off
  RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]
</IfModule>


# ---- 2. Cache do navegador --------------------------------------------------
# Arquivos estáticos do site ficam em cache; o HTML não, para você
# publicar alterações e elas aparecerem na hora.
<IfModule mod_expires.c>
  ExpiresActive On

  ExpiresByType text/css                "access plus 1 year"
  ExpiresByType application/javascript  "access plus 1 year"
  ExpiresByType text/javascript         "access plus 1 year"
  ExpiresByType image/svg+xml           "access plus 1 year"
  ExpiresByType image/jpeg              "access plus 1 year"
  ExpiresByType image/png               "access plus 1 year"
  ExpiresByType image/webp              "access plus 1 year"
  ExpiresByType image/x-icon            "access plus 1 year"
  ExpiresByType font/woff2              "access plus 1 year"

  ExpiresByType text/html               "access plus 0 seconds"
</IfModule>


# ---- 3. Compressão ----------------------------------------------------------
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/plain
  AddOutputFilterByType DEFLATE text/javascript application/javascript
  AddOutputFilterByType DEFLATE application/json image/svg+xml
</IfModule>


# ---- 4. Cabeçalhos de segurança --------------------------------------------
<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"

  # Cache longo para os assets versionados por pasta
  <FilesMatch "\.(css|js|svg|jpg|jpeg|png|webp|woff2)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>


# ---- 5. Página de erro ------------------------------------------------------
ErrorDocument 404 /index.html
